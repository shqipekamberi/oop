#include <iostream>

class File{

    File(char* extention = ".txt", string name = "Test", int capacity = 5)
    {
        this->extension = new char[strlen(extension) + 1];

        this->name = name;
        this->capacity = capacity;
    }
    File(File& other)
    {
        this->extension = new char[strlen(other.extension) + 1];
        strcpy(this->extension, other.extension);
        this->name = other.name;
        this->capacity = other.capacity;
    }
    ~File()
    {
        delete[]extension;
    }

    void print()
    {
        cout << "\nExtension: " << extension << endl;
        cout << "Name: " << name << endl;
        cout << "Capacity: " << capacity << " MB" << endl;
    }
    void input()
    {
        cout << "\nFile extension: ";
        cin >> extension;
        cout << "File name: ";
        cin >> name;
        cout << "Capacity: ";
        cin >> capacity;
    }
};


void printFiles(File files[], int n)
{
    for(int i = 0; i < n; i++)
    {
        if(files[i].extension == ".exe" && files[i].capacity > 1)
        {
            files[i].print();
        }

    }
}
int maxCapacityFile(File files[], int n)
{
    int max = files[0].capacity;
    for(int i = 0; i < n; i++)
    {
        if(files[i].capacity > max)
        {
            max = files[i].capacity;
        }
    }
    return max;
}

int main()
{
    File files[3];
    cout << "Number of files: ";
    int n;
    cin >> n;

    for(int i = 0; i < 3; i++)
    {
        files[i].input();
    }

    cout << "Files that have extension .exe and have capacityy over 1: ";
    printFiles(files, 3);

    cout << "Maximum capacity from given files: " << maxCapacityFile(files, 3) << endl;

    return 0;

}
